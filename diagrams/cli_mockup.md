# Прототип CLI-интерфейса устройства аудита

## Схема взаимодействия

```mermaid
graph TD
    A[Запуск устройства] --> B[Загрузка Kali Linux]
    B --> C[Автозапуск скрипта audit.py]
    C --> D{Главное меню}
    
    D -->|1| E[Сканирование Wi-Fi сетей]
    D -->|2| F[Атака deauth]
    D -->|3| G[Формирование отчета]
    D -->|4| H[Просмотр логов]
    D -->|5| I[Настройки]
    D -->|6| J[Выключение]
    
    E --> K[Вывод списка сетей]
    K --> D
    
    F --> L[Ввод MAC цели]
    L --> M[Подтверждение атаки]
    M --> N[Логирование результата]
    N --> D
    
    G --> O[Выбор формата: PDF/TXT]
    O --> P[Сохранение отчета]
    P --> Q[Вывод пути к файлу]
    Q --> D
    
    H --> R[Отображение журнала аудита]
    R --> D
    
    I --> S{Настройки}
    S -->|1| T[Изменить интерфейс]
    S -->|2| U[Обновить утилиты]
    S -->|3| V[Настроить питание]
    T --> D
    U --> D
    V --> D
    
    J --> W[Завершение работы]
    W --> X[Выключение]
